var out =
  document.getElementById("out");

// PAD.motion / PAD.mic are
// undefined before 0.24.0: keys only
if (PAD.motion) {
  PAD.on("tilt", tilt);
  start(PAD.motion);
}
if (PAD.mic) {
  PAD.on("sound", sound);
  start(PAD.mic);
}

// readings: hold the last value
function tilt(t) {
  out.textContent =
    "x " + t.x.toFixed(2) + " " +
    "y " + t.y.toFixed(2) + " " +
    (t.heading == null ?
      "-" : t.heading);
}
function sound(s) {
  out.textContent =
    "level " + s.level.toFixed(2) +
    " " + (s.pitch || "-");
}

// start() asks the player, then
// answers "on", "denied" or "none";
// it never rejects
function start(sensor) {
  sensor.start().then(function (r) {
    out.textContent = r;
  }, function () {
    out.textContent = "none";
  });
}
// stop when done: the mic holds
// the phone's audio session
function stop(sensor) {
  sensor.stop();
}
